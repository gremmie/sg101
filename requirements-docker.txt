# Requirements for the Dockerized SG101 dev environment -- Python 3.10 / Django
# 4.2. Mirrors the production pins in requirements.txt, plus the dev/test-only
# tools needed to run the suite and dev server in-container.

# --- Runtime (mirror of requirements.txt) ------------------------------------
Django==4.2.30
django-environ==0.4.5  # 12-factor env-var config
# django-tagging is vendored into the repo as `tagging/` -- see the note in
# requirements.txt and in tagging/__init__.py.
pytz==2021.3
# six is a transitive dependency only -- see the note in requirements.txt.
six==1.16.0
# sqlparse is a hard runtime dependency of Django as of 2.2 (and also powers
# debug-toolbar's SQL panel below). Django 4.2 requires >= 0.3.1.
sqlparse==0.5.5

# Celery 5.2 is the last line that supports Python 3.7 (5.3 requires 3.8+). The
# uppercase settings compat is gone in celery 5, so the config moved to the
# CELERY_ namespace -- see sg101/celery.py and sg101/settings.
celery==5.2.7
kombu==5.2.4
billiard==3.6.4.0
amqp==5.1.1
vine==5.0.0
# kombu 5.2's entry-point shim still calls entry_points().get(); importlib-metadata
# 5+ returns a .get()-less EntryPoints, so cap it on the 4.x line (kombu 5.3 fixes
# this but needs Python 3.8+).
importlib-metadata==4.13.0

# mysqlclient is NOT here for Django -- the site runs on PostgreSQL and nothing
# points the ORM back at MySQL. It is here so this image can run
# tools/pgloader/verify.py and verify_data.py, which import MySQLdb to read the
# MySQL source and the converted PostgreSQL side by side; verify.py's header says
# to run it "inside the web container, which has both drivers", and this pin is
# what makes that true. The mysql CLI loads the dump and pgloader converts it --
# only the verification needs the driver. Dropping it leaves a conversion
# unverifiable rather than broken, and neither tool is in the test suite, so
# nothing would go red. See requirements.txt for the longer note.
mysqlclient==1.4.6
# psycopg2 (not psycopg 3): Django gained psycopg 3 support in 4.2, so the
# driver is now a live choice rather than a fixed one -- but swapping it is its
# own change on its own evidence, not a passenger on a version bump, so it stays
# put for now. Built from source (libpq-dev in the Dockerfile), not
# psycopg2-binary, which upstream discourages for anything but a quick trial.
# The 2.9.9 pin dates from the Python 3.7 cap (2.9.10 requires 3.8+), which no
# longer binds.
psycopg2==2.9.9
# redis-py is the client for three things now: the application's own Redis use,
# the Celery broker transport, and Django's built-in Redis cache backend
# (which needs >= 3.0). No django-redis package is involved -- that backend
# has been part of Django since 4.0.
redis==3.5.3

# Pillow 6.2.2 installed and imported fine on 3.10 and then aborted the process
# at interpreter shutdown -- "free(): invalid pointer", exit 134, on both amd64
# and arm64. It is a use-after-free in the C extension's teardown against a
# CPython it predates by two years, and it means any process that imports PIL
# and exits normally dies on a signal: a worker looks like it crashed, a
# management command fails a shell's error check. 9.5.0 is quiet and is the last
# release that still has Image.ANTIALIAS, which 7 call sites use -- replacing
# those is its own change, so this pin stops one version short of needing it.
Pillow==9.5.0
lxml==4.9.4
# bleach 3.1.5 could not import on Python 3.10 (it read Mapping out of
# collections), so it moved with the interpreter. 6.4.0 is the final release --
# upstream archived bleach in 2023 in favor of nh3 -- so this pin is at the end
# of the line rather than behind it.
bleach==6.4.0
html5lib==1.1
Markdown==2.6.11
docutils==0.16
ftfy==5.9
simplejson==3.17.6
python-dateutil==2.8.2
repoze.timeago==0.5
requests==2.27.1

# django-elsewhere is now vendored into the repo (was an editable git dep); the
# boto / google-api-python-client / oauth2client / httplib2 / uritemplate / rsa /
# pyasn1 lines are gone with the stubbed s3 and gcalendar integrations.

# --- Dev/test-only (from requirements_dev.txt) -------------------------------
mock==4.0.3
testfixtures==6.18.5
# debug-toolbar 2.2 caps at Django 3.0; 3.2.x supports Django 2.2-3.2 and stays on
# the Python 3.6+ line. Past its supported Django range here but still loading and
# panelling correctly; the pin moves at the dependency-alignment step. Its SQL
# panel uses sqlparse, pinned in the runtime block.
django-debug-toolbar==3.2.4
# Coverage for the CI report (and local use). <5 keeps a lighter, well-tested
# line that's fine on Python 3.7.
coverage<5
