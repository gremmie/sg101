# Requirements for the Dockerized SG101 dev environment -- Python 3.10 / Django
# 4.2. Mirrors the production pins in requirements.txt, plus the dev/test-only
# tools needed to run the suite and dev server in-container.

# --- Runtime (mirror of requirements.txt) ------------------------------------
Django==4.2.30
django-environ==0.4.5  # 12-factor env-var config
# django-tagging is vendored into the repo as `tagging/` -- see the note in
# requirements.txt and in tagging/__init__.py.
pytz==2021.3
# The IANA timezone database. Required, not incidental -- the base image carries
# no "backward" zone names and most profiles are stored under one. See the note
# in requirements.txt.
tzdata==2026.3
# six is a transitive dependency only, now down to two requirers (html5lib and
# python-dateutil) -- see the note in requirements.txt.
six==1.16.0
# sqlparse is a hard runtime dependency of Django as of 2.2 (and also powers
# debug-toolbar's SQL panel below). Django 4.2 requires >= 0.3.1.
sqlparse==0.5.5

# The uppercase settings compat is gone in celery 5, so the config moved to the
# CELERY_ namespace -- see sg101/celery.py and sg101/settings. The 5.2 line was
# held down by Python 3.7; kombu 5.2 reads its transport registry through
# entry_points().get(), which the standard library removed in 3.12 -- see the
# longer note in requirements.txt.
celery==5.6.3
kombu==5.6.2
billiard==4.2.4
amqp==5.3.1
vine==5.1.0

# mysqlclient is NOT here for Django -- the site runs on PostgreSQL and nothing
# points the ORM back at MySQL. It is here so this image can run
# tools/pgloader/verify.py and verify_data.py, which import MySQLdb to read the
# MySQL source and the converted PostgreSQL side by side; verify.py's header says
# to run it "inside the web container, which has both drivers", and this pin is
# what makes that true. The mysql CLI loads the dump and pgloader converts it --
# only the verification needs the driver. Dropping it leaves a conversion
# unverifiable rather than broken, and neither tool is in the test suite, so
# nothing would go red. See requirements.txt for the longer note.
mysqlclient==1.4.6
# psycopg2, not psycopg 3 -- deferred deliberately, because the pgloader
# verification tools import psycopg2 directly and so adopting psycopg 3 would add
# a second driver rather than replace one. See the longer note in
# requirements.txt. Built from source
# (libpq-dev in the Dockerfile), not psycopg2-binary. The old 2.9.9 pin dated
# from the Python 3.7 cap (2.9.10 requires 3.8+), which no longer binds.
psycopg2==2.9.12
# redis-py is the client for three things now: the application's own Redis use,
# the Celery broker transport, and Django's built-in Redis cache backend
# (which needs >= 3.0). No django-redis package is involved -- that backend
# has been part of Django since 4.0. It moved off 3.5.3 with celery/kombu and
# only because of them -- see requirements.txt for what 3.5.3 silently cost.
redis==6.4.0

# Pillow 10 removed Image.ANTIALIAS; the 7 call sites now name
# Image.Resampling.LANCZOS, the same filter under its own name. The used PIL
# surface and the thumbnail output are both unchanged from 9.5.0 through 12.3.0
# -- see requirements.txt for what was diffed and how.
Pillow==12.3.0
lxml==4.9.4
# bleach 3.1.5 could not import on Python 3.10 (it read Mapping out of
# collections), so it moved with the interpreter. 6.4.0 is the final release --
# upstream archived bleach in 2023 in favor of nh3 -- so this pin is at the end
# of the line rather than behind it.
bleach==6.4.0
html5lib==1.1
# The site's markup engine. The two extensions in core/mdexts/ are written
# against the 3.x extension API (extendMarkdown(md), Registry.register with a
# numeric priority), so they do not load on 2.x.
Markdown==3.10.3
docutils==0.16
ftfy==5.9
simplejson==3.17.6
python-dateutil==2.8.2
repoze.timeago==0.5
requests==2.27.1

# django-elsewhere is now vendored into the repo (was an editable git dep); the
# boto / google-api-python-client / oauth2client / httplib2 / uritemplate / rsa /
# pyasn1 lines are gone with the stubbed s3 and gcalendar integrations.

# --- Dev/test-only (from requirements_dev.txt) -------------------------------
mock==4.0.3
testfixtures==6.18.5
# 6.3.0 is the newest debug-toolbar that still supports Django 4.2 -- 7.0 requires
# Django >= 5.2, so this pin moves again at the 5.2 rung and not before. It had
# been running past its supported Django range since 3.2.x. Its SQL panel uses
# sqlparse, pinned in the runtime block.
django-debug-toolbar==6.3.0
# Coverage for the CI report (and local use). The <5 cap was for Python 3.7.
coverage==7.15.4
