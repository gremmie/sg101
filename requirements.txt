# SG101 runtime requirements -- Python 3.10 / Django 4.2.
#
# Apps whose old external libraries aren't Python-3 ready are temporarily
# stubbed, and their dependencies dropped here:
#   boto (core/s3.py)                       -> Cloudflare R2 / boto3 later
#   google-api-python-client / oauth2client /
#     httplib2 / uritemplate / rsa / pyasn1 (gcalendar) -> modern google-auth later
#
# django-elsewhere used to be an editable git dependency; it's now vendored into
# the repo (the fork is tiny and frozen upstream, and needed Py3 patches), so it
# no longer appears here.

Django==4.2.30
django-environ==0.4.5  # 12-factor env-var config
# django-tagging is no longer installed from PyPI: 0.5.0 is the final release
# (upstream stopped in 2018) and it does not import on Django 4.0, so it is
# vendored into the repo as `tagging/` -- same package name and app label, so
# its migrations and tables are untouched. See tagging/__init__.py.
pytz==2021.3
# six is no longer imported by the application -- it uses the Python 3 standard
# library directly (urllib.parse/request/error, io.StringIO). It is still pulled
# in transitively, so the pin stays for a reproducible build; drop it when the
# last package requiring it is gone or bumped.
six==1.16.0
# sqlparse is a hard runtime dependency of Django as of 2.2 (used internally to
# split/format SQL). It installs automatically with Django; pinned here so the
# build stays reproducible. Django 4.2 raised its floor to 0.3.1, so this is a
# real constraint now and not only a reproducibility pin.
sqlparse==0.5.5

# Celery 5.2 is the last line that supports Python 3.7 (5.3 requires 3.8+). The
# uppercase settings compat is gone in celery 5, so the config moved to the
# CELERY_ namespace -- see sg101/celery.py and sg101/settings.
celery==5.2.7
kombu==5.2.4
billiard==3.6.4.0
amqp==5.1.1
vine==5.0.0
# kombu 5.2's entry-point shim still calls entry_points().get(); importlib-metadata
# 5+ returns a .get()-less EntryPoints, so cap it on the 4.x line (kombu 5.3 fixes
# this but needs Python 3.8+).
importlib-metadata==4.13.0

# mysqlclient is NOT here for Django -- the site runs on PostgreSQL and nothing
# points the ORM back at MySQL. It is here because tools/pgloader/verify.py and
# verify_data.py import MySQLdb: they read the MySQL source and the converted
# PostgreSQL side by side to check that a conversion did not lose or shift
# anything. The mysql CLI loads the dump and pgloader does the converting;
# neither needs this driver, and only the verification does. Removing it would
# leave a conversion unverifiable rather than failing outright -- and since those
# tools are not part of the test suite, nothing would go red. That matters most
# at the production cutover, which converts a fresh snapshot under a maintenance
# window. The 1.4.x line has no wheel, so it builds from source; see the
# Dockerfile for the two apt packages that lets it.
mysqlclient==1.4.6
# psycopg2 (not psycopg 3): Django gained psycopg 3 support in 4.2, so the
# driver is now a live choice rather than a fixed one -- but swapping it is its
# own change on its own evidence, not a passenger on a version bump, so it stays
# put for now. Built from source (libpq-dev in the Dockerfile), not
# psycopg2-binary, which upstream discourages for anything but a quick trial.
# The 2.9.9 pin dates from the Python 3.7 cap (2.9.10 requires 3.8+), which no
# longer binds.
psycopg2==2.9.9
# redis-py is the client for three things now: the application's own Redis use,
# the Celery broker transport, and Django's built-in Redis cache backend
# (which needs >= 3.0). No django-redis package is involved -- that backend
# has been part of Django since 4.0.
redis==3.5.3

# Pillow 6.2.2 installed and imported fine on 3.10 and then aborted the process
# at interpreter shutdown -- "free(): invalid pointer", exit 134, on both amd64
# and arm64. It is a use-after-free in the C extension's teardown against a
# CPython it predates by two years, and it means any process that imports PIL
# and exits normally dies on a signal: a worker looks like it crashed, a
# management command fails a shell's error check. 9.5.0 is quiet and is the last
# release that still has Image.ANTIALIAS, which 7 call sites use -- replacing
# those is its own change, so this pin stops one version short of needing it.
Pillow==9.5.0
lxml==4.9.4
# bleach 3.1.5 could not import on Python 3.10 (it read Mapping out of
# collections), so it moved with the interpreter. 6.4.0 is the final release --
# upstream archived bleach in 2023 in favor of nh3 -- so this pin is at the end
# of the line rather than behind it.
bleach==6.4.0
html5lib==1.1
Markdown==2.6.11
docutils==0.16
ftfy==5.9
simplejson==3.17.6
python-dateutil==2.8.2
repoze.timeago==0.5
requests==2.27.1
